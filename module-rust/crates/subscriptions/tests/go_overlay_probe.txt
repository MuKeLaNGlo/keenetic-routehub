package main

import (
    "encoding/json"
    "os"
    "time"
)

// This test-only entry point runs production functions over synthetic fixtures.
// Go's build overlay supplies this file without changing the original module.
func main() {
    var input struct {
        Op string
        Body string
        Initial string
        Fresh string
        Sources []subscription
    }
    if err := json.NewDecoder(os.Stdin).Decode(&input); err != nil {
        panic(err)
    }
    output := make(map[string]any)
    switch input.Op {
    case "sources":
        saved, err := validateSubscriptions(input.Sources)
        if err != nil {
            output["error"] = err.Error()
            break
        }
        states := make([]subscriptionStatus, 0, len(saved))
        for _, sub := range saved {
            states = append(states, sub.status())
        }
        output["private"] = saved
        output["public"] = publicSubscriptions(saved)
        output["statuses"] = states
    case "merge":
        first, _, err := parseSubscription([]byte(input.Initial))
        if err != nil { panic(err) }
        fresh, _, err := parseSubscription([]byte(input.Fresh))
        if err != nil { panic(err) }
        source := subscription{ID: "example"}
        preview := emptySubscriptionPreview(source.ID)
        old, err := mergeSubscriptionProfiles(plan{}, source, first, &preview)
        if err != nil { panic(err) }
        old[0].Label = "Псевдоним"
        preview = emptySubscriptionPreview(source.ID)
        preview.FetchedAt = time.Unix(0, 0).UTC()
        updated, err := mergeSubscriptionProfiles(plan{Profiles:old,DefaultTarget:old[1].ID}, source, fresh, &preview)
        if err != nil { panic(err) }
        revision, count := subscriptionSnapshot(updated,source.ID)
        output["profiles"], output["preview"] = updated, preview
        output["revision"], output["count"] = revision, count
    default:
        values, incompatible, err := parseSubscription([]byte(input.Body))
        if incompatible == nil { incompatible = []incompatibleSubscriptionRow{} }
        output["incompatible"] = incompatible
        if err != nil { output["error"] = err.Error() } else { output["profiles"] = values }
    }
    if err := json.NewEncoder(os.Stdout).Encode(output); err != nil { panic(err) }
}
